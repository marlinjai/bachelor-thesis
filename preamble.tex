% --- Literaturverzeichnis (Autor-Jahr, biblatex + biber)
% Normen und Rechtsquellen haben keinen Autor und wuerden sonst mit ihrem
% vollen Titel zitiert. Sie tragen daher in references.bib ein shorthand-Feld
% (etwa "EN 338"), das biblatex im Zitat anstelle des Titels ausgibt.
\usepackage[backend=biber, style=authoryear, language=autobib,
            maxcitenames=2, maxbibnames=99,
            uniquename=false, uniquelist=false, natbib=true, dashed=false,
            giveninits=true]{biblatex}
% Ohne diese Zeile zieht biblatex bei Eintraegen ohne Jahr das Abrufdatum als
% Jahresangabe heran. Eine Norm ohne verifiziertes Ausgabejahr erschiene dann
% als Dokument des Abrufjahres. Sie soll stattdessen "o. D." tragen.
\DeclareLabeldate{\field{date}\field{year}\literal{nodate}}
\addbibresource{references.bib}
