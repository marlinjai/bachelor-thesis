% listof=totoc: KOMA adds the figure/table lists to the ToC itself, so we must
% NOT add them manually as well (that produced duplicate headings).
% Le lingue sono opzioni GLOBALI di classe, non opzioni di babel: cleveref
% rileva la lingua solo da qui, altrimenti ricade sull'inglese ("Chapter 2").
\documentclass[oneside,bibliography=totocnumbered,listof=totoc,BCOR=5mm,12pt,
               english,italian,ngerman]{scrbook}

\input{metadata}
\input{preamble}

\title{\thesisTitle}
\author{\thesisAuthor}
\date{\thesisDate}

\begin{document}

\input{frontmatter/titlepage}

% ---------- Vorspann (roemische Nummerierung) ----------
\frontmatter

\tableofcontents

\input{frontmatter/acknowledgments}
\input{frontmatter/abstract-de}
\input{frontmatter/abstract-en}   % Abstract in Zweitsprache

% listof=totoc handles the ToC entries for these two.
\listoffigures
\listoftables

% No code listings in this thesis; re-enable if you add any.
% Under KOMA + listof=totoc, listings registers itself; the heading text is
% renamed to "List of Listings" in preamble.tex.
%\lstlistoflistings

% The `toc' option of glossaries already adds both of these to the ToC.
\printglossary[title=Glossar]
\printglossary[type=\acronymtype,title=Abkuerzungsverzeichnis]

% ---------- Hauptteil (arabische Nummerierung) ----------
\mainmatter

\input{chapters/01-introduction}
\input{chapters/02-background}
\input{chapters/03-literature}
\input{chapters/04-data}
\input{chapters/05-empirical-strategy}
\input{chapters/06-results}
\input{chapters/07-discussion}
\input{chapters/08-conclusion}

% Literaturverzeichnis direkt nach dem Hauptteil (deutsche Konvention).
% heading=bibintoc lascia a biblatex il titolo, la voce nell'indice e la
% testatina: farlo a mano lasciava una testatina errata nelle pagine seguenti.
\printbibliography[heading=bibintoc, title={Literaturverzeichnis}]

% ---------- Anhang ----------
% \clearpage first: without it, \pagenumbering{Roman} takes effect while the
% bibliography's last page is still open, so References would be numbered "I".
\clearpage
\appendix
\pagenumbering{Roman}
\input{backmatter/appendix}

% ---------- Nachspann ----------
\backmatter
\input{backmatter/declaration}

\end{document}
