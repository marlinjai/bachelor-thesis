\newglossaryentry{carbonleakage}{
  name=carbon leakage,
  description={The relocation of emissions-intensive production, or of demand
               for its output, from a jurisdiction with a carbon price to one
               without, such that global emissions fall by less than the
               domestic reduction suggests}
}

\newglossaryentry{embeddedemissions}{
  name=embedded emissions,
  description={The greenhouse gases released during the production of a good,
               attributed to that good when it crosses a border, irrespective
               of where the emissions physically occurred}
}

\newglossaryentry{freeallocation}{
  name=free allocation,
  description={Emission allowances issued at no charge to installations judged
               to be at risk of carbon leakage; under the CBAM regulation these
               are phased out for covered sectors as the levy phases in}
}

\newglossaryentry{resourceshuffling}{
  name=resource shuffling,
  description={Redirecting the cleanest units of an otherwise unchanged
               production portfolio towards the regulated market, so that
               measured import carbon intensity falls without any change in
               actual production emissions}
}

\newglossaryentry{unitvalue}{
  name=unit value,
  description={Trade value divided by trade quantity, used as a proxy for
               price when transaction prices are unobserved}
}
