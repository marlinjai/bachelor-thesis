\newglossaryentry{API}{
  name=Application Programming Interface,
  description={A set of protocols and tools for building software applications}
}

\newglossaryentry{ML}{
  name=Machine Learning,
  description={A subset of artificial intelligence that enables computers to learn
               and improve from experience without being explicitly programmed}
}

\newglossaryentry{SPA}{
  name=Single Page Application,
  description={A web application that renders within a single page without full reloads}
}

\newglossaryentry{CMS}{
  name=Content Management System,
  description={A software system for creating, managing and publishing content
               without requiring technical knowledge}
}

\newglossaryentry{algorithm}{
  name=Algorithm,
  description={A step-by-step procedure or formula for solving a problem}
}
