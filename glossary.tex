\newglossaryentry{carbonleakage}{
  name=rilocalizzazione delle emissioni,
  description={Lo spostamento della produzione ad alta intensità emissiva, o
               della domanda che la riguarda, da una giurisdizione che applica
               un prezzo del carbonio a una che non lo applica, cosicché le
               emissioni globali diminuiscono meno di quanto suggerisca la
               riduzione interna}
}

\newglossaryentry{embeddedemissions}{
  name=emissioni incorporate,
  description={I gas serra rilasciati durante la produzione di un bene e
               attribuiti a quel bene nel momento in cui attraversa la
               frontiera, indipendentemente da dove siano fisicamente avvenute}
}

\newglossaryentry{freeallocation}{
  name=assegnazione gratuita,
  description={Quote di emissione rilasciate a titolo gratuito agli impianti
               ritenuti esposti al rischio di rilocalizzazione delle emissioni;
               ai sensi del regolamento CBAM esse vengono progressivamente
               eliminate per i settori coperti man mano che il prelievo entra
               a regime}
}

\newglossaryentry{resourceshuffling}{
  name=riallocazione contabile della produzione,
  description={La pratica di dirottare verso il mercato regolamentato le unità
               più pulite di un portafoglio produttivo per il resto invariato,
               così che l'intensità emissiva misurata delle importazioni si
               riduca senza alcuna variazione delle emissioni effettive}
}

\newglossaryentry{unitvalue}{
  name=valore unitario,
  description={Il valore degli scambi diviso per la quantità scambiata,
               impiegato come approssimazione del prezzo quando i prezzi di
               transazione non sono osservabili}
}
